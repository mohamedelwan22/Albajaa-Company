import jwt from 'jsonwebtoken';
import bcrypt from 'bcryptjs';
import { NextRequest } from 'next/server';
const SECRET = process.env.JWT_SECRET!;
export const hashPassword = (p: string) => bcrypt.hash(p, 12);
export const verifyPassword = (p: string, h: string) => bcrypt.compare(p, h);
export function generateToken(payload: { userId: string; email: string; role: string }) {
  return jwt.sign(payload, SECRET, { expiresIn: '7d' });
}
export function verifyToken(token: string): any {
  try { return jwt.verify(token, SECRET); } catch { return null; }
}
export function getToken(req: NextRequest): string | null {
  const auth = req.headers.get('authorization');
  if (auth?.startsWith('Bearer ')) return auth.slice(7);
  return req.cookies.get('token')?.value ?? null;
}
export async function getCurrentUser(req: NextRequest) {
  const token = getToken(req);
  if (!token) return null;
  const p = verifyToken(token);
  if (!p) return null;
  return prisma.user.findUnique({ where: { id: p.userId }, include: { company: true } });
}

import { NextResponse } from 'next/server';
import type { NextRequest } from 'next/server';
import { verifyToken } from './lib/auth';

const PUBLIC = ['/api/auth/', '/login', '/', '/_next', '/favicon.ico'];

export function middleware(request: NextRequest) {
  const { pathname } = request.nextUrl;
  if (PUBLIC.some(p => pathname.startsWith(p))) return NextResponse.next();
  const token = request.cookies.get('token')?.value
    ?? request.headers.get('authorization')?.replace('Bearer ', '');
  if (!token) {
    if (pathname.startsWith('/api/')) return NextResponse.json({ error: 'غير مصرح' }, { status: 401 });
    return NextResponse.redirect(new URL('/login', request.url));
  }
  const payload = verifyToken(token);
  if (!payload) {
    if (pathname.startsWith('/api/')) return NextResponse.json({ error: 'جلسة منتهية' }, { status: 401 });
    return NextResponse.redirect(new URL('/login', request.url));
  }
  const headers = new Headers(request.headers);
  headers.set('x-user-id', payload.userId);
  headers.set('x-user-role', payload.role);
  return NextResponse.next({ request: { headers } });
}

import { prisma } from '@/lib/prisma';

export async function GET(req: NextRequest, { params }: { params: { id: string } }) {
  const user = await getCurrentUser(req);
  if (!user) return NextResponse.json({ error: 'غير مصرح' }, { status: 401 });
  const ticket = await prisma.ticket.findUnique({ where: { id: params.id }, include: { company: true } });
  if (!ticket || ticket.companyId !== user.companyId)
    return NextResponse.json({ error: 'غير موجود' }, { status: 404 });
  return NextResponse.json({ ticket });
}

export async function PATCH(req: NextRequest, { params }: { params: { id: string } }) {
  const user = await getCurrentUser(req);
  if (!user) return NextResponse.json({ error: 'غير مصرح' }, { status: 401 });
  const body = await req.json();
  const allowed = ['passengerName','ticketNumber','bookingReference','flightFrom','flightTo',
    'departureDate','departureTime','arrivalDate','arrivalTime','airline','flightNumber',
    'cabinClass','baggageAllowance','gate','price','currency','issueDate','hidePrice'];
  const data: any = {};
  for (const k of allowed) if (k in body) data[k] = body[k];
  const ticket = await prisma.ticket.update({ where: { id: params.id }, data });
  return NextResponse.json({ success: true, ticket });
}

    const d = await res.json();
    if (d.success) { setTicket(d.ticket); setMsg('✅ تم توليد التذكرة'); }
    else setMsg(`❌ ${d.error}`);
    setGenerating(false);
  }

  if (!ticket) return <div className="min-h-screen bg-[#1A1A1A] flex items-center justify-center text-white">جاري التحميل...</div>;

  return (
    <div className="min-h-screen bg-[#1A1A1A] p-6">
      <div className="max-w-4xl mx-auto">
        <div className="flex items-center gap-4 mb-8">
          <Link href="/dashboard" className="text-gray-400 hover:text-white transition-all">← الرئيسية</Link>
          <h1 className="text-2xl font-black text-white">تفاصيل التذكرة</h1>
        </div>

        {msg && <div className="mb-4 p-3 bg-white/10 rounded-xl text-sm text-white">{msg}</div>}

        {/* Raw Text Toggle */}
        {ticket.rawText && (
          <div className="mb-6">
            <button onClick={() => setShowRaw(!showRaw)}
              className="px-4 py-2 border border-[#00AEEF]/50 text-[#00AEEF] rounded-xl text-sm font-bold hover:bg-[#00AEEF]/10 transition-all">
              {showRaw ? 'إخفاء' : 'عرض'} نص التذكرة الأصلي
            </button>
            {showRaw && (
              <div className="mt-3 p-4 bg-white/5 border border-white/10 rounded-xl">
                <p className="text-gray-300 text-xs font-mono whitespace-pre-wrap leading-relaxed max-h-60 overflow-y-auto" dir="ltr">
                  {ticket.rawText}
                </p>
              </div>
            )}
          </div>
        )}

        {/* Form */}
        <div className="bg-white/5 border border-white/10 rounded-2xl p-6 mb-6">
          <div className="flex justify-between items-center mb-6">
            <h2 className="text-white font-bold text-lg">بيانات التذكرة</h2>
            <label className="flex items-center gap-2 cursor-pointer">
              <span className="text-gray-400 text-sm">إخفاء السعر</span>
              <input type="checkbox" checked={form.hidePrice || false}
                onChange={e => setForm({ ...form, hidePrice: e.target.checked })}
                className="w-4 h-4 accent-[#F7931E]"
              />
            </label>
          </div>
          <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
            {FIELDS.map(f => (
              <div key={f.key}>
                <label className="block text-sm text-gray-400 mb-1">{f.label}</label>
                <input
                  type="text"
                  value={form[f.key] || ''}
                  onChange={e => setForm({ ...form, [f.key]: e.target.value })}
                  className="w-full px-4 py-2 bg-white/10 border border-white/20 rounded-xl text-white placeholder-gray-600 focus:outline-none focus:border-[#F7931E] text-sm"
                />
              </div>
            ))}
          </div>
          <div className="mt-6 flex gap-3">
            <button onClick={saveChanges} disabled={saving}
              className="px-6 py-2 bg-[#00AEEF] text-white rounded-xl font-bold text-sm hover:bg-[#0099d6] transition-all disabled:opacity-50">
              {saving ? 'جاري الحفظ...' : 'حفظ التعديلات'}
            </button>
            <button onClick={generate} disabled={generating}
              className="px-6 py-2 bg-[#F7931E] text-white rounded-xl font-bold text-sm hover:bg-[#e8850a] transition-all disabled:opacity-50">
              {generating ? 'جاري التوليد...' : '🎫 توليد التذكرة PDF'}
            </button>
          </div>
        </div>

        {/* Generated PDF */}
        {ticket.generatedFileUrl && (
          <div className="bg-green-500/10 border border-green-500/30 rounded-2xl p-6">
            <h3 className="text-green-400 font-bold mb-4">✅ التذكرة المولّدة جاهزة</h3>
            <div className="flex gap-3">
              <a href={ticket.generatedFileUrl} target="_blank" rel="noopener noreferrer"
                className="px-6 py-2 bg-green-500 text-white rounded-xl font-bold text-sm hover:bg-green-600 transition-all">
                📥 تحميل PDF
              </a>
              <a href={`https://wa.me/?text=${encodeURIComponent(ticket.generatedFileUrl)}`} target="_blank"
                className="px-6 py-2 bg-[#25D366] text-white rounded-xl font-bold text-sm hover:bg-[#20bd5a] transition-all">
                واتساب
              </a>
            </div>
          </div>
        )}
      </div>
    </div>
  );
}
